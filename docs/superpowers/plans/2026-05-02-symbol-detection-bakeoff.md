"""Common types for the symbol-detection bake-off.

All detectors implement Detector ABC and return DetectionResult. Coordinates
are normalized 0-1 fractions of page (BBox.x_pct etc.) so they're DPI-agnostic.
"""

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional


@dataclass(frozen=True)
class BBox:
    """Axis-aligned bounding box in normalized page coordinates (0.0 - 1.0)."""
    x_pct: float
    y_pct: float
    w_pct: float
    h_pct: float

    def __post_init__(self) -> None:
        for name, val in [
            ("x_pct", self.x_pct), ("y_pct", self.y_pct),
            ("w_pct", self.w_pct), ("h_pct", self.h_pct),
        ]:
            if not (0.0 <= val <= 1.0):
                raise ValueError(f"BBox.{name}={val} outside [0, 1]")


@dataclass
class DetectionResult:
    """One detector's output for one (pdf, page, symbol_code) call."""
    count: int                       # Reported count (authoritative for ranking)
    bboxes: list[BBox]               # Optional spatial detail; some pipelines
                                     # only count and leave this empty
    runtime_s: float                 # Wall time for this single detect() call
    cost_usd: float                  # API cost ($ for this call), 0 for local
    errors: list[str] = field(default_factory=list)  # Non-fatal warnings
    notes: str = ""                  # Free-form for human inspection


class Detector(ABC):
    """Common interface for every pipeline.

    Lifecycle:
        det = ConcretePipeline()
        det.setup(pdf_path)        # one-time per PDF: extract legend, prerender, etc.
        result = det.detect(pdf_path, page_nr=3, symbol_code="rozete")
        # ... call detect() many times for different (page, symbol) combos
        det.teardown()             # optional cleanup
    """
    name: str = "base"               # Subclass overrides

    @abstractmethod
    def setup(self, pdf_path: Path) -> None:
        """One-time prep per PDF. May extract legend, render pages, build index."""

    @abstractmethod
    def detect(self, pdf_path: Path, page_nr: int, symbol_code: str) -> DetectionResult:
        """Detect all instances of `symbol_code` on `page_nr` of `pdf_path`."""

    def teardown(self) -> None:  # noqa: B027 - optional override
        """Cleanup (close caches, kill subprocesses). Default no-op."""
```

```python
"""Render PDF pages to grayscale numpy arrays, caching to disk.

Cache key: (pdf size + mtime + page_nr + dpi). File size+mtime is cheap and
catches "I re-exported the PDF" without hashing megabytes.
"""

from __future__ import annotations
import os
import io
import hashlib
from pathlib import Path
from typing import Tuple

import numpy as np
from PIL import Image
from pdf2image import convert_from_path
import pikepdf


def _cache_dir() -> Path:
    base = os.environ.get(
        "DETECTOR_BENCH_RENDER_CACHE",
        str(Path(__file__).resolve().parent.parent.parent / "detector_bench" / "render_cache"),
    )
    p = Path(base)
    p.mkdir(parents=True, exist_ok=True)
    return p


def _cache_key(pdf_path: Path, page_nr: int, dpi: int) -> str:
    stat = pdf_path.stat()
    raw = f"{pdf_path.name}|{stat.st_size}|{int(stat.st_mtime)}|p{page_nr}|d{dpi}"
    return hashlib.sha1(raw.encode("utf-8")).hexdigest()[:20]


def render_page(pdf_path: Path, page_nr: int, dpi: int = 200) -> np.ndarray:
    """Returns the page rendered as a uint8 grayscale numpy array.

    page_nr is 1-based (matches PDF convention).
    DPI 200 is enough for symbol detection; higher is slower with no quality
    gain since the source is already a flattened raster.
    """
    pdf_path = Path(pdf_path)
    if not pdf_path.exists():
        raise FileNotFoundError(f"PDF not found: {pdf_path}")
    cache_path = _cache_dir() / f"{_cache_key(pdf_path, page_nr, dpi)}.png"
    if cache_path.exists():
        img = Image.open(cache_path).convert("L")
        return np.asarray(img, dtype=np.uint8)
    pages = convert_from_path(
        str(pdf_path), dpi=dpi, first_page=page_nr, last_page=page_nr,
        fmt="png", grayscale=True,
    )
    if not pages:
        raise ValueError(f"page {page_nr} not found in {pdf_path}")
    img = pages[0]
    img.save(cache_path, "PNG")
    return np.asarray(img, dtype=np.uint8)


def get_page_dims_pdf(pdf_path: Path, page_nr: int) -> Tuple[float, float]:
    """Returns (width, height) in PDF points for the given page.

    Honors page rotation (rotated 90°/270° pages report dims as the user sees
    them, not as stored).
    """
    pdf_path = Path(pdf_path)
    with pikepdf.open(str(pdf_path)) as pdf:
        if page_nr < 1 or page_nr > len(pdf.pages):
            raise IndexError(f"page {page_nr} out of range (1..{len(pdf.pages)})")
        page = pdf.pages[page_nr - 1]
        mb = page.mediabox
        w = float(mb[2]) - float(mb[0])
        h = float(mb[3]) - float(mb[1])
        rot = int(page.get("/Rotate", 0)) % 360
        if rot in (90, 270):
            w, h = h, w
        return w, h
```

```python
"""Ground truth (human-confirmed symbol counts) per (pdf, page).

YAML schema:
    pdf_name: projektas_3_breziniai.pdf
    page_nr: 23
    dpi: 200                       # render DPI when ground truth was authored
    symbols:
      rozete: 47
      jungtukas: 19
      led_strip: 8
    confirmed_at: '2026-05-02T18:00:00Z'
    confirmed_by: user
    notes: electrical layout, 1st floor
"""

from __future__ import annotations
from dataclasses import dataclass, field
from pathlib import Path
import yaml


@dataclass
class GroundTruth:
    pdf_name: str
    page_nr: int
    dpi: int
    symbols: dict[str, int]
    confirmed_at: str
    confirmed_by: str
    notes: str = ""


@dataclass
class PageGroundTruth:
    """Convenience alias used in benchmark.py."""
    gt: GroundTruth


def save_ground_truth(gt: GroundTruth, path: Path) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    data = {
        "pdf_name": gt.pdf_name,
        "page_nr": gt.page_nr,
        "dpi": gt.dpi,
        "symbols": dict(gt.symbols),
        "confirmed_at": gt.confirmed_at,
        "confirmed_by": gt.confirmed_by,
        "notes": gt.notes,
    }
    with path.open("w", encoding="utf-8") as f:
        yaml.safe_dump(data, f, sort_keys=False, allow_unicode=True)


def load_ground_truth(path: Path) -> GroundTruth:
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"ground truth not found: {path}")
    with path.open("r", encoding="utf-8") as f:
        data = yaml.safe_load(f)
    syms = data.get("symbols") or {}
    if not isinstance(syms, dict):
        raise ValueError("symbols must be a mapping {symbol_code: int}")
    for k, v in syms.items():
        if not isinstance(v, int):
            raise ValueError(f"symbols.{k}={v!r} must be int")
    return GroundTruth(
        pdf_name=data["pdf_name"],
        page_nr=int(data["page_nr"]),
        dpi=int(data.get("dpi", 200)),
        symbols=syms,
        confirmed_at=str(data.get("confirmed_at", "")),
        confirmed_by=str(data.get("confirmed_by", "")),
        notes=str(data.get("notes", "")),
    )
```

"""Auto-extract symbol legends from a PDF.

Given a PDF + the page numbers where SUTARTINIAI ŽENKLAI appears, ask Claude
Sonnet 4.6 (vision-capable) to read the legend block and return structured:

    [
      {
        "symbol_code": "iš.1",
        "description_lt": "išviršinis lubinis šviestuvas",
        "category": "lighting",
        "glyph_bbox_pct": [x, y, w, h],   # location of the GLYPH inside the page
        "color_hint": "#RRGGBB" | null,
      },
      ...
    ]

Then crops each glyph_bbox_pct out of the rendered page and saves as PNG
under legends/<pdf_name>/<symbol_code>.png — these become templates for
Pipeline 3 (cv2.matchTemplate) and Pipeline 4 (hybrid).
"""

from __future__ import annotations
import json
import os
import base64
from pathlib import Path
from io import BytesIO

from PIL import Image
import anthropic

from detector_bench.core.pdf_loader import render_page

CLAUDE_MODEL = "claude-sonnet-4-6"

LEGEND_PROMPT = """You are reading a Lithuanian construction-drawing legend block.

Identify every distinct symbol shown in the SUTARTINIAI ŽENKLAI / SUTARTINIAI ŽYMĖJIMAI section
of this page. For each symbol, return:

- symbol_code: the short code (e.g. "iš.1", "pš.1", "led.1", "rozete", "jungtukas").
  Lowercase, replace spaces with underscores. If the legend uses a Lithuanian word
  (e.g. "rozetė", "jungtukas") with no code, use the word itself.
- description_lt: the Lithuanian description shown next to the symbol.
- category: one of "lighting" | "electrical" | "plumbing" | "hvac" | "finish" | "other"
- glyph_bbox_pct: [x, y, w, h] normalized 0-1 fractions of THIS PAGE indicating
  where the symbol GLYPH (not the description text) is drawn. Tight crop.
- color_hint: dominant glyph color as #RRGGBB hex if obviously colored, else null.

Return ONLY valid JSON — a list of objects, no prose, no markdown fence.
If you cannot find a legend on this page, return [].
"""


def _img_to_base64_png(img: Image.Image) -> str:
    buf = BytesIO()
    img.save(buf, format="PNG")
    return base64.b64encode(buf.getvalue()).decode("ascii")


def extract_legend_from_page(pdf_path: Path, page_nr: int) -> list[dict]:
    """Returns list of symbol dicts (see module docstring). Empty if no legend."""
    if not os.environ.get("ANTHROPIC_API_KEY"):
        raise RuntimeError("ANTHROPIC_API_KEY not set")
    arr = render_page(Path(pdf_path), page_nr, dpi=200)
    img = Image.fromarray(arr)
    b64 = _img_to_base64_png(img)
    client = anthropic.Anthropic()
    resp = client.messages.create(
        model=CLAUDE_MODEL,
        max_tokens=4000,
        messages=[
            {
                "role": "user",
                "content": [
                    {"type": "image", "source": {"type": "base64", "media_type": "image/png", "data": b64}},
                    {"type": "text", "text": LEGEND_PROMPT},
                ],
            }
        ],
    )
    text = resp.content[0].text.strip()
    # Tolerate accidental code fences
    if text.startswith("```"):
        text = text.split("```", 2)[1]
        if text.startswith("json\n"):
            text = text[5:]
        text = text.rstrip("`").strip()
    try:
        items = json.loads(text)
    except json.JSONDecodeError as e:
        raise ValueError(f"Claude returned non-JSON: {e}\n--- raw ---\n{text[:500]}")
    if not isinstance(items, list):
        raise ValueError(f"expected list, got {type(items).__name__}")
    return items


def crop_glyph_templates(
    pdf_path: Path, page_nr: int, items: list[dict], out_dir: Path,
) -> dict[str, Path]:
    """Crop each glyph_bbox_pct out of the rendered page; save as PNG.

    Returns mapping {symbol_code: png_path}.
    """
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    arr = render_page(Path(pdf_path), page_nr, dpi=200)
    h, w = arr.shape
    img = Image.fromarray(arr)
    out: dict[str, Path] = {}
    for item in items:
        bbox = item.get("glyph_bbox_pct")
        code = item.get("symbol_code")
        if not bbox or not code or len(bbox) != 4:
            continue
        x, y, ww, hh = bbox
        left = max(0, int(x * w))
        top = max(0, int(y * h))
        right = min(w, int((x + ww) * w))
        bottom = min(h, int((y + hh) * h))
        if right <= left or bottom <= top:
            continue
        crop = img.crop((left, top, right, bottom))
        # Sanitize filename
        safe = "".join(c if c.isalnum() or c in "._-" else "_" for c in code)
        out_path = out_dir / f"{safe}.png"
        crop.save(out_path, "PNG")
        out[code] = out_path
    return out
```

For symbols whose architects label with words ("Rozetė", "Jungtukas",
"Lubinis šviestuvas"), count occurrences of those Lithuanian terms in
the PDF text layer on the given page.

We use a per-symbol-code keyword map. The map is heuristic — built from
common Lithuanian architectural vocabulary plus the legend-extractor's
description_lt field as a fallback.

Strength: zero cost, milliseconds. Strength on P1/VT (text-heavy).
Weakness: P3 has minimal annotations — text-driven returns 0 there.
"""

from __future__ import annotations
import time
import re
from pathlib import Path

import pdfplumber

from detector_bench.core.types import Detector, DetectionResult, BBox

# Per-symbol-code text patterns. Generic mapping of category → keywords.
# Specific symbol codes (iš.1, pš.1) fall through to substring match in the legend's description_lt.
SYMBOL_KEYWORDS: dict[str, list[str]] = {
    "rozete":    [r"rozet[ėeė]",       r"rozet"],
    "jungtukas": [r"jungtuk",          r"jungiklis"],
    "led_strip": [r"led\s*juost",      r"led\s*strip"],
    "lubinis_sviestuvas": [r"lubinis\s+šviestuv",  r"luba\s+šviestuv"],
    "pakabinamas_sviestuvas": [r"pakabinam[oa]s\s+šviestuv"],
    "vandens_isvedimas": [r"vandens\s+išved",  r"sifon"],
    "ventiliatorius":    [r"ventiliator"],
}


def _keywords_for(symbol_code: str, legend_description_lt: str | None = None) -> list[str]:
    """Best-effort keyword list for a symbol code. Falls back to description tokens."""
    sc = symbol_code.lower().strip()
    if sc in SYMBOL_KEYWORDS:
        return SYMBOL_KEYWORDS[sc]
    # Try substring match against description_lt (e.g. "iš.1" -> "išviršinis lubinis šviestuvas")
    if legend_description_lt:
        # Take the most distinctive lowercase word ≥4 chars
        words = re.findall(r"[a-zA-ZąčęėįšųūžĄČĘĖĮŠŲŪŽ]{4,}", legend_description_lt.lower())
        if words:
            return [re.escape(w) for w in words[:2]]
    # Last fallback — match the literal code
    return [re.escape(sc)]


class TextDrivenDetector(Detector):
    name = "p2_text_driven"

    def __init__(self) -> None:
        self._legend: dict[str, str] = {}  # symbol_code → description_lt

    def setup(self, pdf_path: Path) -> None:
        # Per-PDF legend lives at legends/<pdf_stem>.json
        import json
        legend_json = Path(f"/Users/ignasgaldikas/samatu-ai/legends/{Path(pdf_path).stem}.json")
        if legend_json.exists():
            for it in json.loads(legend_json.read_text()):
                self._legend[it["symbol_code"]] = it.get("description_lt", "")

    def detect(self, pdf_path: Path, page_nr: int, symbol_code: str) -> DetectionResult:
        t0 = time.time()
        keywords = _keywords_for(symbol_code, self._legend.get(symbol_code))
        try:
            with pdfplumber.open(str(pdf_path)) as pdf:
                if page_nr < 1 or page_nr > len(pdf.pages):
                    return DetectionResult(
                        count=0, bboxes=[], runtime_s=time.time() - t0,
                        cost_usd=0.0, errors=[f"page {page_nr} out of range"],
                    )
                page = pdf.pages[page_nr - 1]
                text = (page.extract_text() or "").lower()
            count = 0
            for kw in keywords:
                count += len(re.findall(kw, text, re.IGNORECASE))
            return DetectionResult(
                count=count, bboxes=[],
                runtime_s=time.time() - t0, cost_usd=0.0, errors=[],
                notes=f"keywords matched: {keywords}",
            )
        except Exception as e:
            return DetectionResult(
                count=0, bboxes=[], runtime_s=time.time() - t0,
                cost_usd=0.0, errors=[str(e)], notes="failed",
            )

```python
"""Pipeline 3 — cv2.matchTemplate from auto-extracted legend glyphs.

For each (page, symbol_code):
  1. Load template at legends/<pdf_stem>/<symbol_code>.png (cropped legend glyph)
  2. Render the page at the same DPI (200)
  3. Multi-scale, multi-rotation template match
  4. Non-maximum suppression on matches above threshold
  5. Return count + bboxes
"""

from __future__ import annotations
import time
from pathlib import Path

import numpy as np
import cv2

from detector_bench.core.types import Detector, DetectionResult, BBox
from detector_bench.core.pdf_loader import render_page

# Tuneable — affects precision/recall heavily.
DEFAULT_THRESHOLD = 0.75
SCALES = [0.85, 1.0, 1.15]              # tolerance for scale variation
ROTATIONS = [0, 90, 180, 270]           # symbols often rotated 90° on plumbing/HVAC
NMS_IOU_THRESHOLD = 0.3                 # suppress overlapping detections


def _rotate(img: np.ndarray, angle: int) -> np.ndarray:
    if angle == 0:
        return img
    if angle == 90:
        return np.rot90(img, k=1)
    if angle == 180:
        return np.rot90(img, k=2)
    if angle == 270:
        return np.rot90(img, k=3)
    raise ValueError(f"unsupported rotation: {angle}")


def _nms(boxes: list[tuple[int, int, int, int, float]], iou_thresh: float) -> list[tuple[int, int, int, int, float]]:
    """Non-max suppression. Boxes: list of (x, y, w, h, score)."""
    if not boxes:
        return []
    boxes_np = np.array([[b[0], b[1], b[0] + b[2], b[1] + b[3]] for b in boxes], dtype=np.float32)
    scores = np.array([b[4] for b in boxes])
    keep = []
    idxs = np.argsort(-scores)
    while len(idxs) > 0:
        i = idxs[0]
        keep.append(i)
        if len(idxs) == 1:
            break
        rest = idxs[1:]
        xx1 = np.maximum(boxes_np[i, 0], boxes_np[rest, 0])
        yy1 = np.maximum(boxes_np[i, 1], boxes_np[rest, 1])
        xx2 = np.minimum(boxes_np[i, 2], boxes_np[rest, 2])
        yy2 = np.minimum(boxes_np[i, 3], boxes_np[rest, 3])
        w = np.maximum(0, xx2 - xx1)
        h = np.maximum(0, yy2 - yy1)
        inter = w * h
        a_i = (boxes_np[i, 2] - boxes_np[i, 0]) * (boxes_np[i, 3] - boxes_np[i, 1])
        a_rest = (boxes_np[rest, 2] - boxes_np[rest, 0]) * (boxes_np[rest, 3] - boxes_np[rest, 1])
        iou = inter / (a_i + a_rest - inter + 1e-6)
        idxs = rest[iou < iou_thresh]
    return [boxes[k] for k in keep]


class TemplateMatchDetector(Detector):
    name = "p3_template_match"

    def __init__(self, threshold: float = DEFAULT_THRESHOLD) -> None:
        self.threshold = threshold

    def setup(self, pdf_path: Path) -> None:
        self._pdf_stem = Path(pdf_path).stem

    def detect(self, pdf_path: Path, page_nr: int, symbol_code: str) -> DetectionResult:
        t0 = time.time()
        # Sanitize symbol_code → filename (must match crop_glyph_templates)
        safe = "".join(c if c.isalnum() or c in "._-" else "_" for c in symbol_code)
        template_path = Path(f"/Users/ignasgaldikas/samatu-ai/legends/{self._pdf_stem}/{safe}.png")
        if not template_path.exists():
            return DetectionResult(
                count=0, bboxes=[], runtime_s=time.time() - t0,
                cost_usd=0.0, errors=[f"template not found: {template_path}"],
                notes="missing template",
            )

        page = render_page(pdf_path, page_nr, dpi=200)
        template = cv2.imread(str(template_path), cv2.IMREAD_GRAYSCALE)
        if template is None or template.size == 0:
            return DetectionResult(
                count=0, bboxes=[], runtime_s=time.time() - t0,
                cost_usd=0.0, errors=["template empty"], notes="bad template",
            )

        h_p, w_p = page.shape
        all_matches: list[tuple[int, int, int, int, float]] = []
        for scale in SCALES:
            for angle in ROTATIONS:
                rot = _rotate(template, angle)
                t_h = max(8, int(rot.shape[0] * scale))
                t_w = max(8, int(rot.shape[1] * scale))
                if t_h >= h_p or t_w >= w_p:
                    continue
                resized = cv2.resize(rot, (t_w, t_h), interpolation=cv2.INTER_AREA)
                res = cv2.matchTemplate(page, resized, cv2.TM_CCOEFF_NORMED)
                ys, xs = np.where(res >= self.threshold)
                for x, y in zip(xs, ys):
                    score = float(res[y, x])
                    all_matches.append((int(x), int(y), t_w, t_h, score))

        kept = _nms(all_matches, NMS_IOU_THRESHOLD)
        bboxes = [
            BBox(
                x_pct=b[0] / w_p, y_pct=b[1] / h_p,
                w_pct=b[2] / w_p, h_pct=b[3] / h_p,
            )
            for b in kept
        ]
        return DetectionResult(
            count=len(kept), bboxes=bboxes,
            runtime_s=time.time() - t0, cost_usd=0.0, errors=[],
            notes=f"threshold={self.threshold} scales={SCALES} rots={ROTATIONS} raw={len(all_matches)} nms={len(kept)}",
        )
```

Use text positions from pdfplumber as candidate locations, then template-match in a small neighborhood to confirm symbol type. Best of P2 + P3 for text-annotated PDFs.

```python
"""Pipeline 4 — Hybrid: text positions as anchors, template match as confirmer.

For each text occurrence of a keyword (e.g. "Rozetė") on the page:
  1. Get its bbox from pdfplumber
  2. Crop a small neighborhood around it (extend by ~150 px)
  3. Run template match within that neighborhood
  4. If template matches → count this anchor as a confirmed symbol

This catches symbols that have both a label AND a glyph, with high precision.
Symbols with NO text label fall back to template-match-on-whole-page.
"""

from __future__ import annotations
import time
import re
from pathlib import Path

import numpy as np
import cv2
import pdfplumber

from detector_bench.core.types import Detector, DetectionResult, BBox
from detector_bench.core.pdf_loader import render_page, get_page_dims_pdf
from detector_bench.detectors.p2_text_driven import _keywords_for
from detector_bench.detectors.p3_template_match import (
    TemplateMatchDetector, SCALES, ROTATIONS, _rotate, _nms,
)

NEIGHBORHOOD_PX = 200          # search radius around each text anchor


class HybridDetector(Detector):
    name = "p4_hybrid"

    def __init__(self, threshold: float = 0.65) -> None:
        # Lower threshold than pure template — text confirms presence,
        # template just confirms type.
        self.threshold = threshold

    def setup(self, pdf_path: Path) -> None:
        self._pdf_stem = Path(pdf_path).stem
        import json
        legend_json = Path(f"/Users/ignasgaldikas/samatu-ai/legends/{self._pdf_stem}.json")
        self._legend: dict[str, str] = {}
        if legend_json.exists():
            for it in json.loads(legend_json.read_text()):
                self._legend[it["symbol_code"]] = it.get("description_lt", "")

    def detect(self, pdf_path: Path, page_nr: int, symbol_code: str) -> DetectionResult:
        t0 = time.time()
        keywords = _keywords_for(symbol_code, self._legend.get(symbol_code))
        safe = "".join(c if c.isalnum() or c in "._-" else "_" for c in symbol_code)
        template_path = Path(f"/Users/ignasgaldikas/samatu-ai/legends/{self._pdf_stem}/{safe}.png")
        page_arr = render_page(pdf_path, page_nr, dpi=200)
        h_p, w_p = page_arr.shape

        try:
            pdf_w_pt, pdf_h_pt = get_page_dims_pdf(pdf_path, page_nr)
        except Exception as e:
            return DetectionResult(
                count=0, bboxes=[], runtime_s=time.time() - t0,
                cost_usd=0.0, errors=[f"page dims: {e}"], notes="failed",
            )
        # Convert PDF points → pixels at 200 DPI
        pt_to_px_x = w_p / pdf_w_pt
        pt_to_px_y = h_p / pdf_h_pt

        # Step 1: text anchors via pdfplumber
        anchors: list[tuple[int, int]] = []  # (x_px, y_px) of each text occurrence
        try:
            with pdfplumber.open(str(pdf_path)) as pdf:
                page = pdf.pages[page_nr - 1]
                words = page.extract_words(extra_attrs=["x0", "x1", "top", "bottom"])
                for w in words:
                    txt = (w.get("text") or "").lower()
                    for kw in keywords:
                        if re.search(kw, txt, re.IGNORECASE):
                            cx = (w["x0"] + w["x1"]) / 2
                            cy = (w["top"] + w["bottom"]) / 2
                            anchors.append((int(cx * pt_to_px_x), int(cy * pt_to_px_y)))
                            break
        except Exception as e:
            return DetectionResult(
                count=0, bboxes=[], runtime_s=time.time() - t0,
                cost_usd=0.0, errors=[f"pdfplumber: {e}"], notes="failed",
            )

        # Step 2: if template exists, confirm each anchor
        confirmed_bboxes: list[BBox] = []
        if template_path.exists() and anchors:
            template = cv2.imread(str(template_path), cv2.IMREAD_GRAYSCALE)
            if template is not None and template.size > 0:
                for (ax, ay) in anchors:
                    x0 = max(0, ax - NEIGHBORHOOD_PX)
                    x1 = min(w_p, ax + NEIGHBORHOOD_PX)
                    y0 = max(0, ay - NEIGHBORHOOD_PX)
                    y1 = min(h_p, ay + NEIGHBORHOOD_PX)
                    nbh = page_arr[y0:y1, x0:x1]
                    if nbh.size == 0:
                        continue
                    found = False
                    for scale in SCALES:
                        for angle in ROTATIONS:
                            rot = _rotate(template, angle)
                            t_h = max(8, int(rot.shape[0] * scale))
                            t_w = max(8, int(rot.shape[1] * scale))
                            if t_h >= nbh.shape[0] or t_w >= nbh.shape[1]:
                                continue
                            resized = cv2.resize(rot, (t_w, t_h), interpolation=cv2.INTER_AREA)
                            res = cv2.matchTemplate(nbh, resized, cv2.TM_CCOEFF_NORMED)
                            if res.max() >= self.threshold:
                                ys, xs = np.where(res == res.max())
                                yy, xx = int(ys[0]), int(xs[0])
                                bx = (x0 + xx) / w_p
                                by = (y0 + yy) / h_p
                                bw = t_w / w_p
                                bh = t_h / h_p
                                confirmed_bboxes.append(BBox(bx, by, bw, bh))
                                found = True
                                break
                        if found:
                            break

                # Fallback if no anchors confirmed: full-page template match
                if not confirmed_bboxes:
                    fallback = TemplateMatchDetector(threshold=self.threshold + 0.10)
                    fallback.setup(pdf_path)
                    return fallback.detect(pdf_path, page_nr, symbol_code)

                return DetectionResult(
                    count=len(confirmed_bboxes), bboxes=confirmed_bboxes,
                    runtime_s=time.time() - t0, cost_usd=0.0, errors=[],
                    notes=f"text anchors={len(anchors)} confirmed={len(confirmed_bboxes)}",
                )

        # No template: use text count alone
        return DetectionResult(
            count=len(anchors), bboxes=[],
            runtime_s=time.time() - t0, cost_usd=0.0, errors=[],
            notes=f"text-only (no template): anchors={len(anchors)}",
        )
```

### Task 12: Pipeline 5 — GroundingDINO via Replicate

Open-vocabulary grounding-vision: prompt with the legend description, get bboxes back.

```python
"""Pipeline 5 — GroundingDINO open-vocabulary detection via Replicate.

Sends the rendered page + a text prompt ("electrical outlet symbol") to a
pretrained GroundingDINO model on Replicate, gets bboxes back. Designed
specifically for "find all instances of [text description] in this image",
which is exactly our problem.

Cost: ~$0.005 per call on Replicate (varies by model; we use a public
checkpoint). Runtime: 2-5s per call.
"""

from __future__ import annotations
import os
import time
import base64
from io import BytesIO
from pathlib import Path

from PIL import Image
import replicate

from detector_bench.core.types import Detector, DetectionResult, BBox
from detector_bench.core.pdf_loader import render_page

# A widely-used GroundingDINO endpoint on Replicate. If this becomes unavailable,
# the model name needs swapping to a current alternative (any open-vocabulary
# detector with image+text input that returns bboxes works).
MODEL = "adirik/grounding-dino:efd10a8ddc57ea28773327e881ce95e20cc1d734c589f7dd01d2036921ed9e1f"
COST_PER_CALL_USD = 0.005


class GroundingDinoDetector(Detector):
    name = "p5_grounding_dino"

    def __init__(self, threshold: float = 0.30) -> None:
        self.threshold = threshold

    def setup(self, pdf_path: Path) -> None:
        self._pdf_stem = Path(pdf_path).stem
        import json
        legend_json = Path(f"/Users/ignasgaldikas/samatu-ai/legends/{self._pdf_stem}.json")
        self._legend: dict[str, str] = {}
        if legend_json.exists():
            for it in json.loads(legend_json.read_text()):
                self._legend[it["symbol_code"]] = it.get("description_lt", "")

    def _prompt_for(self, symbol_code: str) -> str:
        # GroundingDINO works best with English noun phrases.
        # Map common LT codes → English descriptions.
        lt = self._legend.get(symbol_code, "")
        # Translate common LT terms to grounded English noun phrases
        if "rozet" in (lt + symbol_code).lower():
            return "electrical outlet . electrical socket . wall outlet"
        if "jungtuk" in (lt + symbol_code).lower() or "jungiklis" in (lt + symbol_code).lower():
            return "light switch . wall switch"
        if "lubinis" in lt.lower() or "luba" in lt.lower():
            return "ceiling light fixture . overhead luminaire"
        if "led" in (lt + symbol_code).lower():
            return "LED strip . linear LED light"
        if "ventilator" in (lt + symbol_code).lower():
            return "ventilation fan . exhaust fan"
        # Last resort: literal symbol code
        return f"{symbol_code} symbol"

    def detect(self, pdf_path: Path, page_nr: int, symbol_code: str) -> DetectionResult:
        t0 = time.time()
        if not os.environ.get("REPLICATE_API_TOKEN"):
            return DetectionResult(
                count=0, bboxes=[], runtime_s=0.0, cost_usd=0.0,
                errors=["REPLICATE_API_TOKEN not set"], notes="skipped",
            )
        try:
            arr = render_page(pdf_path, page_nr, dpi=200)
            img = Image.fromarray(arr)
            buf = BytesIO()
            img.save(buf, format="PNG")
            data_uri = "data:image/png;base64," + base64.b64encode(buf.getvalue()).decode("ascii")
            prompt = self._prompt_for(symbol_code)

            output = replicate.run(
                MODEL,
                input={
                    "image": data_uri,
                    "query": prompt,
                    "box_threshold": self.threshold,
                    "text_threshold": 0.20,
                },
            )
            # Output shape varies by model; common shape:
            # { "detections": [{"box": [x1,y1,x2,y2], "label": "...", "confidence": 0.5}, ...] }
            detections = []
            if isinstance(output, dict):
                detections = output.get("detections") or output.get("boxes") or []
            elif isinstance(output, list):
                detections = output
            h_p, w_p = arr.shape
            bboxes: list[BBox] = []
            for det in detections:
                box = det.get("box") if isinstance(det, dict) else det
                if not box or len(box) != 4:
                    continue
                x1, y1, x2, y2 = [float(v) for v in box]
                # Some endpoints return absolute pixels, others normalized.
                # Detect by max value; if any > 1.0 assume pixels.
                if max(x1, y1, x2, y2) > 1.0:
                    x1 /= w_p; x2 /= w_p; y1 /= h_p; y2 /= h_p
                if x2 <= x1 or y2 <= y1:
                    continue
                bboxes.append(BBox(
                    x_pct=max(0, min(1, x1)),
                    y_pct=max(0, min(1, y1)),
                    w_pct=max(0, min(1, x2 - x1)),
                    h_pct=max(0, min(1, y2 - y1)),
                ))
            return DetectionResult(
                count=len(bboxes), bboxes=bboxes,
                runtime_s=time.time() - t0, cost_usd=COST_PER_CALL_USD,
                errors=[], notes=f"prompt='{prompt}' raw_detections={len(detections)}",
            )
        except Exception as e:

### Task 13: Pipeline 6 — Claude Sonnet tool-orchestrated

Claude reads the legend, then orchestrates a deterministic count tool (template match) per symbol. Vision is the orchestrator; CV is the counter.

**Files:**
- Create: `detector_bench/detectors/p6_claude_tool_orch.py`

- [ ] **Step 13.1: Implement**

Create `detector_bench/detectors/p6_claude_tool_orch.py`:

```python
"""Pipeline 6 — Claude Sonnet 4.6 with tool use, orchestrating a CV counter.

Claude reads the legend page, then for each symbol calls a tool we provide:
    count_via_template(symbol_code, page_nr) → int

The tool internally runs cv2.matchTemplate (Pipeline 3 logic). Claude's value:
deciding which symbol is which and providing context-aware threshold tuning
("this is a small icon, use higher threshold; this is a large badge, use
lower threshold"). For our benchmark we let it use a default threshold and
just return the deterministic count.
"""

from __future__ import annotations
import os
import time
import json
import base64
from io import BytesIO
from pathlib import Path

import anthropic
from PIL import Image

from detector_bench.core.types import Detector, DetectionResult, BBox
from detector_bench.core.pdf_loader import render_page
from detector_bench.detectors.p3_template_match import TemplateMatchDetector

CLAUDE_MODEL = "claude-sonnet-4-6"
COST_PER_CALL_USD = 0.02  # Higher than baseline due to tool roundtrips


class ClaudeToolOrchestratedDetector(Detector):
    name = "p6_claude_tool_orch"

    def __init__(self) -> None:
        self._template_det = TemplateMatchDetector(threshold=0.75)

    def setup(self, pdf_path: Path) -> None:
        self._template_det.setup(pdf_path)

    def detect(self, pdf_path: Path, page_nr: int, symbol_code: str) -> DetectionResult:
        t0 = time.time()
        if not os.environ.get("ANTHROPIC_API_KEY"):
            return DetectionResult(
                count=0, bboxes=[], runtime_s=0.0, cost_usd=0.0,
                errors=["ANTHROPIC_API_KEY not set"], notes="skipped",
            )

        try:
            arr = render_page(pdf_path, page_nr, dpi=200)
            img = Image.fromarray(arr)
            buf = BytesIO()
            img.save(buf, format="PNG")
            b64 = base64.b64encode(buf.getvalue()).decode("ascii")

            tools = [{
                "name": "count_via_template",
                "description": "Count instances of a symbol on the current page via deterministic OpenCV template match. Returns integer count.",
                "input_schema": {
                    "type": "object",
                    "properties": {
                        "symbol_code": {"type": "string", "description": "Symbol code from the legend"},
                        "threshold": {"type": "number", "description": "Match threshold 0-1 (default 0.75)"},
                    },
                    "required": ["symbol_code"],
                },
            }]

            client = anthropic.Anthropic()
            messages = [{
                "role": "user",
                "content": [
                    {"type": "image", "source": {"type": "base64", "media_type": "image/png", "data": b64}},
                    {"type": "text", "text": (
                        f"You are counting symbol '{symbol_code}' on this construction drawing. "
                        f"Use the count_via_template tool to get a deterministic count. "
                        f"Adjust threshold if you think the default 0.75 is too strict (small/varied symbols) "
                        f"or too lenient (common-looking glyphs). Return ONLY the final integer count via the tool result; "
                        f"no prose."
                    )},
                ],
            }]

            # Tool-use loop
            count = 0
            for _ in range(3):  # max 3 iterations
                resp = client.messages.create(
                    model=CLAUDE_MODEL,
                    max_tokens=2000,
                    tools=tools,
                    messages=messages,
                )
                if resp.stop_reason == "tool_use":
                    tool_blocks = [b for b in resp.content if b.type == "tool_use"]
                    if not tool_blocks:
                        break
                    tu = tool_blocks[0]
                    args = tu.input or {}
                    sym = args.get("symbol_code", symbol_code)
                    thr = float(args.get("threshold", 0.75))
                    self._template_det.threshold = thr
                    sub = self._template_det.detect(pdf_path, page_nr, sym)
                    count = sub.count
                    messages.append({"role": "assistant", "content": resp.content})
                    messages.append({
                        "role": "user",
                        "content": [{
                            "type": "tool_result",
                            "tool_use_id": tu.id,
                            "content": str(sub.count),
                        }],
                    })
                else:
                    # End of conversation; final count was set in last tool_result
                    break

            return DetectionResult(
                count=count, bboxes=[],  # Tool doesn't propagate bboxes back
                runtime_s=time.time() - t0, cost_usd=COST_PER_CALL_USD,
                errors=[],
                notes="claude orchestrated template match",
            )
        except Exception as e:
            return DetectionResult(
                count=0, bboxes=[], runtime_s=time.time() - t0,
                cost_usd=COST_PER_CALL_USD, errors=[str(e)], notes="failed",
            )
```

"""Run every detector against every ground truth (page, symbol).

Output: results/<date>-bakeoff.md with side-by-side comparison.
"""

from __future__ import annotations
import os
import time
from datetime import datetime
from pathlib import Path

from detector_bench.core.types import Detector, DetectionResult
from detector_bench.core.ground_truth import GroundTruth, load_ground_truth

from detector_bench.detectors.p1_baseline_vision import BaselineVisionDetector
from detector_bench.detectors.p2_text_driven import TextDrivenDetector
from detector_bench.detectors.p3_template_match import TemplateMatchDetector

from detector_bench.detectors.p6_claude_tool_orch import ClaudeToolOrchestratedDetector


REPO = Path("/Users/ignasgaldikas/samatu-ai")
GT_DIR = REPO / "ground_truth"
RESULTS_DIR = REPO / "results"


def _accuracy(predicted: int, truth: int) -> float:
    """1.0 = exact; 0.5 = off by 50% of truth; 0.0 = ≥ truth wrong."""
    if truth == 0:
        return 1.0 if predicted == 0 else 0.0
    err = abs(predicted - truth) / truth
    return max(0.0, 1.0 - err)


def _pdf_path_for(name: str) -> Path:
    if name.startswith("projektas_1"):
        return REPO / "training-data/projektas_1/projektas_1_breziniai.pdf"
    if name.startswith("projektas_3"):
        return REPO / "training-data/projektas_3/projektas_3_breziniai.pdf"
    if "Visi techniniai" in name or name.startswith("visi_techniniai"):
        return Path("/Users/ignasgaldikas/Desktop/Visi techniniai 01.28.pdf")
    raise ValueError(f"unknown PDF: {name}")


def main() -> None:
    # Load env
    env_path = REPO / "backend" / ".env"
    if env_path.exists():
        for line in env_path.read_text().splitlines():
            if "=" in line and not line.strip().startswith("#"):
                k, v = line.split("=", 1)
                os.environ.setdefault(k.strip(), v.strip())

    detectors: list[Detector] = [
        BaselineVisionDetector(),
        TextDrivenDetector(),
        TemplateMatchDetector(),
        HybridDetector(),
        GroundingDinoDetector(),
        ClaudeToolOrchestratedDetector(),
    ]

    gt_files = sorted(GT_DIR.glob("*.yaml"))
    if not gt_files:
        print(f"No ground truth in {GT_DIR}. Run build_ground_truth.py first.")
        return

    # gt_path → GroundTruth + pdf_path
    bench: list[tuple[Path, GroundTruth]] = []
    for p in gt_files:
        gt = load_ground_truth(p)
        bench.append((_pdf_path_for(gt.pdf_name), gt))

    # results[(detector_name, gt_path, symbol_code)] = (predicted, runtime_s, cost, errors)
    results: dict[tuple[str, str, str], tuple[int, float, float, str]] = {}

    for det in detectors:
        seen_pdfs: set[Path] = set()
        for pdf_path, gt in bench:
            if pdf_path not in seen_pdfs:
                try:
                    det.setup(pdf_path)
                    seen_pdfs.add(pdf_path)
                except Exception as e:
                    print(f"  [{det.name}] setup failed for {pdf_path.name}: {e}")
                    continue
            for symbol_code, _truth in gt.symbols.items():
                try:
                    r = det.detect(pdf_path, gt.page_nr, symbol_code)
                except Exception as e:
                    r = DetectionResult(0, [], 0.0, 0.0, [str(e)])
                key = (det.name, gt.confirmed_at + "|" + gt.pdf_name + "|p" + str(gt.page_nr), symbol_code)
                err_summary = "; ".join(r.errors[:1])[:80]
                results[key] = (r.count, r.runtime_s, r.cost_usd, err_summary)
                print(f"  [{det.name}] {gt.pdf_name} p{gt.page_nr} {symbol_code}: pred={r.count} truth={_truth} ({r.runtime_s:.1f}s)")

    # Build report
    RESULTS_DIR.mkdir(exist_ok=True)
    report_path = RESULTS_DIR / f"{datetime.now():%Y-%m-%d}-bakeoff.md"
    lines: list[str] = []
    lines.append(f"# Symbol-detection bake-off — {datetime.now():%Y-%m-%d %H:%M}\n")
    lines.append(f"Pipelines tested: {len(detectors)}  ·  Ground-truth pages: {len(bench)}\n")

    # Per-detector aggregate accuracy
    lines.append("## Aggregate accuracy across all (page, symbol) pairs\n")
    lines.append("| Detector | Mean accuracy | Total runtime | Total cost | Errors |")
    lines.append("|---|---|---|---|---|")
    for det in detectors:
        accs = []
        total_t = 0.0
        total_c = 0.0
        n_err = 0
        for (pdf_path, gt) in bench:
            for sym, truth in gt.symbols.items():
                if truth < 0:  # skipped during ground truth
                    continue
                key = (det.name, gt.confirmed_at + "|" + gt.pdf_name + "|p" + str(gt.page_nr), sym)
                if key not in results:
                    continue
                pred, rt, cost, err = results[key]
                accs.append(_accuracy(pred, truth))
                total_t += rt
                total_c += cost
                if err:
                    n_err += 1
        mean = sum(accs) / max(1, len(accs))
        lines.append(f"| `{det.name}` | {mean*100:.1f}% | {total_t:.1f}s | ${total_c:.2f} | {n_err} |")

    # Per-PDF breakdown
    lines.append("\n## Per-PDF accuracy\n")
    lines.append("| Detector | " + " | ".join(set(gt.pdf_name for _, gt in bench)) + " |")
    lines.append("|---" * (1 + len(set(gt.pdf_name for _, gt in bench))) + "|")

    pdf_groups: dict[str, list[GroundTruth]] = {}
    for _, gt in bench:
        pdf_groups.setdefault(gt.pdf_name, []).append(gt)

    for det in detectors:
        row = [f"`{det.name}`"]
        for pdf_name in pdf_groups:
            accs = []
            for gt in pdf_groups[pdf_name]:
                for sym, truth in gt.symbols.items():
                    if truth < 0:
                        continue
                    key = (det.name, gt.confirmed_at + "|" + gt.pdf_name + "|p" + str(gt.page_nr), sym)
                    if key not in results:
                        continue
                    pred, *_ = results[key]
                    accs.append(_accuracy(pred, truth))
            mean = sum(accs) / max(1, len(accs))
            row.append(f"{mean*100:.1f}%")
        lines.append("| " + " | ".join(row) + " |")

    # Per-symbol detail (verbose)
    lines.append("\n## Per-page, per-symbol detail\n")
    for pdf_path, gt in bench:
        lines.append(f"### {gt.pdf_name} — page {gt.page_nr}")
        lines.append(f"_{gt.notes}_\n")
        lines.append("| Symbol | Truth | " + " | ".join(d.name for d in detectors) + " |")
        lines.append("|---" * (2 + len(detectors)) + "|")
        for sym, truth in sorted(gt.symbols.items()):
            row = [f"`{sym}`", str(truth)]
            for det in detectors:
                key = (det.name, gt.confirmed_at + "|" + gt.pdf_name + "|p" + str(gt.page_nr), sym)
                if key not in results:
                    row.append("—")
                    continue
                pred, rt, cost, err = results[key]
                if err:
                    row.append(f"⚠️{pred}")
                else:
                    row.append(f"{pred} ({_accuracy(pred, truth)*100:.0f}%)")
            lines.append("| " + " | ".join(row) + " |")
        lines.append("")

    report_path.write_text("\n".join(lines))
    print(f"\nReport: {report_path}")


if __name__ == "__main__":
    main()
